วิธีการทำงาน

ระบบจะส่งข้อมูลผ่าน Google Apps Script Web App ไปยัง Google Sheets ของท่าน โดยข้อมูลจะถูกบันทึกเป็นแถวในชีต สามารถดู แก้ไข และแชร์ได้แบบ Real-time

คู่มือการตั้งค่า (5 ขั้นตอน)

1

ไปที่ sheets.google.com สร้างชีตใหม่ และตั้งชื่อหัวแถวดังนี้:

3

คัดลอกโค้ด Apps Script

ลบโค้ดเดิมทั้งหมด แล้ววางโค้ดด้านล่างนี้ลงไป

/**
 * ระบบอุทธรณ์ - Google Apps Script Backend
 * ภาควิชาเทคโนโลยีและสื่อสารการศึกษา
 */

const SHEET_NAME = 'etcuthon'; // ชื่อชีต (เปลี่ยนได้ตามต้องการ)

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  try {
    const params = e.parameter || {};
    let postData = {};
    
    if (e.postData) {
      try {
        postData = JSON.parse(e.postData.contents);
      } catch (err) {
        postData = {};
      }
    }
    
    const action = params.action || postData.action;
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);
    
    if (!sheet) {
      return jsonResponse({ success: false, error: 'ไม่พบชีต: ' + SHEET_NAME });
    }
    
    switch(action) {
      case 'read':
        return readAll(sheet);
      case 'create':
        return createRecord(sheet, postData);
      case 'update':
        return updateRecord(sheet, postData);
      case 'delete':
        return deleteRecord(sheet, postData);
      default:
        return jsonResponse({ success: false, error: 'action ไม่ถูกต้อง' });
    }
  } catch (err) {
    return jsonResponse({ success: false, error: err.toString() });
  }
}

function readAll(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) {
    return jsonResponse({ success: true, data: [] });
  }
  
  const headers = data[0];
  const records = [];
  
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[0]) continue; // ข้ามแถวว่าง
    const record = {};
    headers.forEach((h, idx) => {
      record[h] = row[idx];
    });
    records.push(record);
  }
  
  return jsonResponse({ success: true, data: records });
}

function createRecord(sheet, data) {
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const row = headers.map(h => data[h] || '');
  sheet.appendRow(row);
  return jsonResponse({ success: true, message: 'สร้างข้อมูลสำเร็จ' });
}

function updateRecord(sheet, data) {
  const allData = sheet.getDataRange().getValues();
  const headers = allData[0];
  const idCol = headers.indexOf('id') + 1;
  
  if (!idCol) {
    return jsonResponse({ success: false, error: 'ไม่พบคอลัมน์ id' });
  }
  
  for (let i = 1; i < allData.length; i++) {
    if (String(allData[i][idCol - 1]) === String(data.id)) {
      headers.forEach((h, idx) => {
        if (data[h] !== undefined) {
          sheet.getRange(i + 1, idx + 1).setValue(data[h]);
        }
      });
      return jsonResponse({ success: true, message: 'อัปเดตสำเร็จ' });
    }
  }
  
  return jsonResponse({ success: false, error: 'ไม่พบข้อมูล id: ' + data.id });
}

function deleteRecord(sheet, data) {
  const allData = sheet.getDataRange().getValues();
  const headers = allData[0];
  const idCol = headers.indexOf('id') + 1;
  
  if (!idCol) {
    return jsonResponse({ success: false, error: 'ไม่พบคอลัมน์ id' });
  }
  
  for (let i = 1; i < allData.length; i++) {
    if (String(allData[i][idCol - 1]) === String(data.id)) {
      sheet.deleteRow(i + 1);
      return jsonResponse({ success: true, message: 'ลบสำเร็จ' });
    }
  }
  
  return jsonResponse({ success: false, error: 'ไม่พบข้อมูล' });
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
